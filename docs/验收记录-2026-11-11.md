## 实际执行方式

以下命令均在 Mac 隔离目录的 `source/` 执行。覆盖文件为镜像设置上述版本／提交号，并为 Playwright 挂载 `../evidence:/evidence:Z`；随机测试管理员凭据只保存在隔离目录内，不进入仓库。

```sh
make test-backend DEV_COMPOSE='podman compose -p netassets-acceptance-dev-20261111 -f compose.dev.yaml'
podman compose -p netassets-acceptance-dev-20261111 -f compose.dev.yaml run --rm --no-deps web sh -c 'npm ci && npm test -- --run --maxWorkers=2 && npm run build'
make build COMPOSE='podman compose -p netassets-acceptance-20261111 -f compose.yaml -f ../image.compose.yaml'

compose() {
  podman compose --env-file ../acceptance.env -p netassets-acceptance-e2e-20261111 -f compose.e2e.yaml -f ../e2e.compose.yaml "$@"
}
compose up -d --no-build app
compose run --rm playwright sh -c 'npm ci'
compose run --rm playwright sh -c 'PLAYWRIGHT_JSON_OUTPUT_FILE=/evidence/core.json npx playwright test tests/security-finding-editor.spec.ts tests/security-finding-editor-live.spec.ts tests/security-overview-live.spec.ts tests/security-issues-1-3.spec.ts tests/role-boundaries.spec.ts tests/role-based-operational-flow.spec.ts --workers=1 --reporter=line,json --output=/workspace/test-results/acceptance-core'
compose run --rm playwright sh -c 'PLAYWRIGHT_JSON_OUTPUT_FILE=/evidence/toolbar.json npx playwright test --config=toolbar-layout.config.ts --workers=1 --reporter=line,json --output=/workspace/test-results/acceptance-toolbar'
compose run --rm playwright sh -c 'PLAYWRIGHT_JSON_OUTPUT_FILE=/evidence/remediation.json npx playwright test tests/remediation-workspace.spec.ts --workers=1 --reporter=line,json --output=/workspace/test-results/acceptance-remediation'
```
